    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(100) NOT NULL,
    emp_salary DECIMAL(10, 2) NOT NULL,
      emp_city VARCHAR(100) NOT NULL
);

select emp_city, count(*) as cnt
from employees
group by emp_city
order by emp_city


select emp_city, count(emp_id) as cnt
from employees
group by emp_city
having count(emp_id)>1
    
SELECT EMP_CITY, COUNT(EMP_ID) AS CNT
FROM EMPLOYEES
GROUP BY EMP_CITY
HAVING COUNT(EMP_ID)>1



SELECT EMP_CITY, Min(EMP_SALARY) AS MAX_SAL
FROM EMPLOYEES
GROUP BY EMP_CITY
order by emp_city

SELECT EMP_CITY, AVG(EMP_SALARY)::NUMERIC(10,1) AS MAX_SAL
FROM EMPLOYEES
GROUP BY EMP_CITY
